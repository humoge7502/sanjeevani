# ---- M2 optimizer (optional at runtime, present by default) ------------------
# PuLP drives the multi-objective recovery program in backend/m2/optimizer.py and
# bundles the CBC solver binary. It is OPTIONAL ON PURPOSE: if the import or the
# solve fails, the optimizer falls back to exact enumeration (which is optimal for
# this network's five-consignment plans) and records why in its trace. A missing
# solver degrades the *explanation*, never the answer.
pulp==4.2.0

# SimPy drives the deterministic discrete-event cross-check in backend/m2/des.py
# (chokepoint berths, closure windows, queue waits). It schedules a deterministic
# event heap -- no sampling, no seed -- so the DES's outputs stay byte-reproducible
# like everything else in the audit chain. Pure Python, no transitive dependencies.
simpy==4.1.2

# NOT here, deliberately:
#   numpy / scipy / pandas / scikit-learn  ->  requirements-ml.txt (training only)
#   pyjwt                                  ->  present; see the note below
#
# PyJWT powers the optional authentication layer (backend/security/tokens.py) and
# IS a runtime dependency once SANJEEVANI_AUTH_ENABLED=true. It is small, pure
# Python and has no transitive dependencies.
PyJWT==2.15.1

# Prometheus metrics exposition for /metrics (backend/observability/metrics.py).
